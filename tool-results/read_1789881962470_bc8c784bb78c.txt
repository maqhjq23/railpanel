    64→    64→export default function FileExplorer({ socket, serverId }: { socket: any; serverId: string }) {
    65→    65→  const [cwd, setCwd] = useState('.')
    66→    66→  const [entries, setEntries] = useState<FileEntry[]>([])
    67→    67→  const [loading, setLoading] = useState(true)
    68→    68→  const [dialog, setDialog] = useState<DialogMode>({ kind: 'none' })
    69→    69→  const [inputValue, setInputValue] = useState('')
    70→    70→  const [editContent, setEditContent] = useState('')
    71→    71→  const [origContent, setOrigContent] = useState('') // buat deteksi dirty
    72→    72→  const [confirmDiscard, setConfirmDiscard] = useState(false)
    73→    73→  const [busy, setBusy] = useState(false)
    74→    74→  const [uploading, setUploading] = useState(false)
    75→    75→  const [selMode, setSelMode] = useState<'none' | 'sel'>('none')
    76→    76→  const [sel, setSel] = useState<Set<string>>(new Set())
    77→    77→  const [gutStart, setGutStart] = useState(0)
    78→    78→  const [gutCount, setGutCount] = useState(60)
    79→    79→  const uploadRef = useRef<HTMLInputElement>(null)
    80→    80→  const inputRef = useRef<HTMLInputElement>(null)
    81→    81→  const editRef = useRef<HTMLTextAreaElement>(null)
    82→    82→  const gutRef = useRef<HTMLDivElement>(null)
    83→    83→  const { toast } = useToast()
    84→    84→
    85→    85→  const load = useCallback(
    86→    86→    async (dir: string = cwd) => {
    87→    87→      setLoading(true)
    88→    88→      try {
    89→    89→        const res = await rpc<{ entries: FileEntry[] }>(socket, 'files:list', { id: serverId, path: dir })
    90→    90→        setEntries(res.entries)
    91→    91→      } catch (e) {
    92→    92→        toast({ title: 'Gagal load file', description: (e as Error).message, variant: 'destructive' })
    93→    93→      } finally {
    94→    94→        setLoading(false)
    95→    95→      }
    96→    96→    },
    97→    97→    [socket, serverId, cwd, toast],
    98→    98→  )
    99→    99→
   100→   100→  useEffect(() => {
   101→   101→    load(cwd)
   102→   102→  }, [cwd])
   103→   103→
   104→   104→  const dlgFrom = 'from' in dialog ? dialog.from : undefined
   105→   105→
   106→   106→  // auto-focus: Radix dialog fokusnya ke tombol close X duluan, jadi kita
   107→   107→  // paksa fokus manual ke input/textarea setelah dialog kebuka.
   108→   108→  useEffect(() => {
   109→   109→    const k = dialog.kind
   110→   110→    if (k === 'none') return
   111→   111→    const t = setTimeout(() => {
   112→   112→      if (k === 'edit') {
   113→   113→        editRef.current?.focus()
   114→   114→      } else if (k === 'newFile' || k === 'newFolder' || k === 'rename' || k === 'zipOut' || k === 'moveTo') {
   115→   115→        inputRef.current?.focus()
   116→   116→        if (k === 'rename') inputRef.current?.select() // nama lama ke-highlight semua
   117→   117→      }
   118→   118→    }, 120)
   119→   119→    return () => clearTimeout(t)
   120→   120→  }, [dialog.kind, dlgFrom])
   121→   121→
   122→   122→  function join(dir: string, name: string) {
   123→   123→    return dir === '.' ? name : `${dir}/${name}`
   124→   124→  }
   125→   125→
   126→   126→  // pindah folder SELALU lewat sini biar mode select ke-reset (gak boleh di effect body)
   127→   127→  function go(dir: string) {
   128→   128→    setSelMode('none')
   129→   129→    setSel(new Set())
   130→   130→    setCwd(dir)
   131→   131→  }
   132→   132→
   133→   133→  function enterSelect(mode: 'none' | 'sel') {
   134→   134→    setSel(new Set())
   135→   135→    setSelMode(mode)
   136→   136→  }
   137→   137→
   138→   138→  function toggleSel(p: string) {
   139→   139→    setSel((prev) => {
   140→   140→      const next = new Set(prev)
   141→   141→      if (next.has(p)) next.delete(p)
   142→   142→      else next.add(p)
   143→   143→      return next
   144→   144→    })
   145→   145→  }
   146→   146→
   147→   147→  function isZip(e: FileEntry) {
   148→   148→    return e.type === 'file' && /\.zip$/i.test(e.name)
   149→   149→  }
   150→   150→
   151→   151→  function isZipPath(p: string) {
   152→   152→    return /\.zip$/i.test(p.split('/').pop() || '')
   153→   153→  }
   154→   154→
   155→   155→  function openFile(p: string) {
   156→   156→    rpc<{ content: string }>(socket, 'files:read', { id: serverId, path: p })
   157→   157→      .then((res) => {
   158→   158→        setInputValue('')
   159→   159→        setEditContent(res.content)
   160→   160→        setOrigContent(res.content)
   161→   161→        setGutStart(0)
   162→   162→        setGutCount(60)
   163→   163→        setDialog({ kind: 'edit', from: p })
   164→   164→      })
   165→   165→      .catch((e) => toast({ title: 'Gagal baca file', description: (e as Error).message, variant: 'destructive' }))
   166→   166→  }
   167→   167→
   168→   168→  // tutup dialog utama — kalau editor dirty, minta konfirmasi dulu
   169→   169→  function closeMain() {
   170→   170→    if (dialog.kind === 'edit' && editContent !== origContent) {
   171→   171→      setConfirmDiscard(true)
   172→   172→    } else {
   173→   173→      setDialog({ kind: 'none' })
   174→   174→    }
   175→   175→  }
   176→   176→
   177→   177→  async function submitDialog() {
   178→   178→    setBusy(true)
   179→   179→    try {
   180→   180→      if (dialog.kind === 'newFile') {
   181→   181→        await rpc(socket, 'files:write', { id: serverId, path: join(cwd, inputValue), content: '' })
   182→   182→      } else if (dialog.kind === 'newFolder') {
   183→   183→        await rpc(socket, 'files:mkdir', { id: serverId, path: join(cwd, inputValue) })
   184→   184→      } else if (dialog.kind === 'rename') {
   185→   185→        await rpc(socket, 'files:rename', { id: serverId, from: dialog.from, to: join(cwd, inputValue) })
   186→   186→      } else if (dialog.kind === 'delete') {
   187→   187→        await rpc(socket, 'files:delete', { id: serverId, path: dialog.from })
   188→   188→      } else if (dialog.kind === 'edit') {
   189→   189→        await rpc(socket, 'files:write', { id: serverId, path: dialog.from, content: editContent })
   190→   190→      } else if (dialog.kind === 'extract') {
   191→   191→        const targets = dialog.from
   192→   192→          ? [dialog.from]
   193→   193→          : Array.from(sel).filter((p) => isZipPath(p))
   194→   194→        let okN = 0
   195→   195→        let lastErr = ''
   196→   196→        for (const t of targets) {
   197→   197→          try {
   198→   198→            await rpc(socket, 'files:extract', { id: serverId, path: t })
   199→   199→            okN++
   200→   200→          } catch (err) {
   201→   201→            lastErr = (err as Error).message
   202→   202→          }
   203→   203→        }
   204→   204→        if (okN === 0) throw new Error(lastErr || 'Extract gagal')
   205→   205→        if (okN < targets.length) {
   206→   206→          toast({ title: `Extract sebagian (${okN}/${targets.length})`, description: lastErr, variant: 'destructive' })
   207→   207→        } else {
   208→   208→          toast({
   209→   209→            title: 'Extract sukses',
   210→   210→            description: targets.length === 1 ? targets[0].split('/').pop() : `${okN} arsip`,
   211→   211→          })
   212→   212→        }
   213→   213→        if (!dialog.from) enterSelect('none')
   214→   214→      } else if (dialog.kind === 'delSel') {
   215→   215→        const targets = Array.from(sel)
   216→   216→        let okN = 0
   217→   217→        let lastErr = ''
   218→   218→        for (const p of targets) {
   219→   219→          try {
   220→   220→            await rpc(socket, 'files:delete', { id: serverId, path: p })
   221→   221→            okN++
   222→   222→          } catch (err) {
   223→   223→            lastErr = (err as Error).message
   224→   224→          }
   225→   225→        }
   226→   226→        if (okN < targets.length) {
   227→   227→          toast({ title: `Hapus sebagian (${okN}/${targets.length})`, description: lastErr, variant: 'destructive' })
   228→   228→        }
   229→   229→        enterSelect('none')
   230→   230→      } else if (dialog.kind === 'zipOut') {
   231→   231→        await rpc(socket, 'files:zip', { id: serverId, items: Array.from(sel), out: join(cwd, inputValue) })
   232→   232→        enterSelect('none') // reset mode + seleksi
   233→   233→      } else if (dialog.kind === 'moveTo') {
   234→   234→        const dest = inputValue.trim() === '' ? '.' : inputValue.trim()
   235→   235→        await rpc(socket, 'files:move', { id: serverId, items: Array.from(sel), dest })
   236→   236→        enterSelect('none')
   237→   237→      }
   238→   238→      setDialog({ kind: 'none' })
   239→   239→      await load()
   240→   240→    } catch (e) {
   241→   241→      toast({ title: 'Gagal', description: (e as Error).message, variant: 'destructive' })
   242→   242→    } finally {
   243→   243→      setBusy(false)
   244→   244→    }
   245→   245→  }
   246→   246→
   247→   247→  async function onUpload(files: FileList | null) {
   248→   248→    if (!files || files.length === 0) return
   249→   249→    setUploading(true)
   250→   250→    try {
   251→   251→      const fd = new FormData()
   252→   252→      fd.append('server', serverId)
   253→   253→      fd.append('path', cwd)
   254→   254→      for (const f of Array.from(files)) fd.append('files', f)
   255→   255→      const res = await fetch('/api/files/upload', { method: 'POST', body: fd })
   256→   256→      const data = await res.json()
   257→   257→      if (!res.ok || !data.ok) throw new Error(data.error || 'Upload gagal')
   258→   258→      toast({ title: 'Upload sukses', description: `${data.saved} file tersimpan` })
   259→   259→      await load()
   260→   260→    } catch (e) {
   261→   261→      toast({ title: 'Upload gagal', description: (e as Error).message, variant: 'destructive' })
   262→   262→    } finally {
   263→   263→      setUploading(false)
   264→   264→      if (uploadRef.current) uploadRef.current.value = ''
   265→   265→    }
   266→   266→  }

   268→   268→  // sinkron gutter nomor baris ke scroll textarea + update rentang nomor yang dirender

   275→   275→    setGutCount((c) => (c === count ? c : c))
   276→   276→  }
   277→   277→
   278→   278→  const crumbs = cwd === '.' ? [] : cwd.split('/')
   279→   279→  const selecting = selMode !== 'none'
   280→   280→  const selPaths = Array.from(sel)
   281→   281→  const selZip = selPaths.filter((p) => isZipPath(p))
   282→   282→  const lineCount = dialog.kind === 'edit' ? Math.max(1, editContent.split('\n').length) : 1
   283→   283→
   284→   284→  return (
   285→   285→    <div className="space-y-3">
   286→   286→      {/* breadcrumb */}
   287→   287→      <div className="flex items-center gap-2">

   293→   293→          disabled={cwd === '.'}

   295→   295→          <ChevronRight className="h-4 w-4 rotate-180" />
   296→   296→        </Button>
   297→   297→        <div className="min-w-0 flex-1 overflow-x-auto font-mono text-sm text-zinc-400 [scrollbar-width:none] [&::-webkit-scrollbar]:hidden">
   298→   298→          <button className="hover:text-emerald-400" onClick={() => go('.')}>/</button>
   299→   299→          {crumbs.map((c, i) => (
   300→   300→            <span key={i}>
   301→   301→              {' / '}
   302→   302→              <button
   303→   303→                className="hover:text-emerald-400"
   304→   304→                onClick={() => go(crumbs.slice(0, i + 1).join('/'))}
   305→   305→              >
   306→   306→                {c}
   307→   307→              </button>
   308→   308→            </span>
   309→   309→          ))}
   310→   310→        </div>
   311→   311→      </div>
   312→   312→
   313→   313→      {/* toolbar "laci": bisa digeser horizontal kalau gak muat */}
   314→   314→      <div className="overflow-x-auto [scrollbar-width:thin] [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-zinc-700 [&::-webkit-scrollbar]:h-1.5">
   315→   315→        <div className="flex w-max gap-2">
   316→   316→          <Button variant="outline" size="sm" className={btnBar} onClick={() => { setInputValue(''); setDialog({ kind: 'newFile' }) }}>
   317→   317→            <FilePlus2 className="h-4 w-4" /> File
   318→   318→          </Button>
   319→   319→          <Button variant="outline" size="sm" className={btnBar} onClick={() => { setInputValue(''); setDialog({ kind: 'newFolder' }) }}>
   320→   320→            <FolderPlus className="h-4 w-4" /> Folder
   321→   321→          </Button>
   322→   322→          <Button variant="outline" size="sm" className={btnBar} disabled={uploading} onClick={() => uploadRef.current?.click()}>
   323→   323→            {uploading ? <Loader2 className="h-4 w-4 animate-spin" /> : <Upload className="h-4 w-4" />} Upload
   324→   324→          </Button>
   325→   325→          <Button
   326→   326→            variant={selecting ? 'default' : 'outline'}
   327→   327→            size="sm"
   328→   328→            className={selecting ? btnSel : btnBar}
   329→   329→            onClick={() => (selecting ? enterSelect('none') : enterSelect('sel'))}
   330→   330→          >
   331→   331→            <ListChecks className="h-4 w-4" /> Pilih
   332→   332→          </Button>
   333→   333→          <Button
   334→   334→            variant={selecting ? 'default' : 'outline'}
   335→   335→            size="sm"
   336→   336→            className={selecting ? btnSel : btnBar}
   337→   337→            onClick={() => (selecting ? enterSelect('none') : enterSelect('sel'))}
   338→   338→          >
   339→   339→            <Archive className="h-4 w-4" /> Zip
   340→   340→          </Button>
   341→   341→          <Button
   342→   342→            variant={selecting ? 'default' : 'outline'}
   343→   343→            size="sm"
   344→   344→            className={selecting ? btnSel : btnBar}
   345→   345→            onClick={() => (selecting ? enterSelect('none') : enterSelect('sel'))}
   346→   346→          >
   347→   347→            <FolderInput className="h-4 w-4" /> Move
   348→   348→          </Button>
   349→   349→          <input
   350→   350→            ref={uploadRef}
   351→   351→            type="file"

   354→   354→            onChange={(e) => onUpload(e.target.files)}

   356→   356→        </div>
   357→   357→      </div>
   358→   358→
   359→   359→      {/* bar aksi mode select: Kompres / Move / Hapus / Extract / Batal */}
   360→   360→      {selecting && (
   361→   361→        <div className="flex flex-wrap items-center gap-2 rounded-lg border border-emerald-900/70 bg-emerald-950/30 px-3 py-2">
   362→   362→          <span className="text-xs font-medium text-emerald-300">
   363→   363→            {sel.size} dipilih{selZip.length > 0 ? ` · ${selZip.length} zip` : ''}
   364→   364→          </span>
   365→   365→          <div className="ml-auto flex flex-wrap items-center gap-2">
   366→   366→            <Button
   367→   367→              variant="outline"
   368→   368→              size="sm"
   369→   369→              className="h-8 border-zinc-800 bg-zinc-950"
   370→   370→              disabled={sel.size === 0 || busy}
   371→   371→              onClick={() => { setInputValue('arsip.zip'); setDialog({ kind: 'zipOut' }) }}
   372→   372→            >
   373→   373→              <Archive className="mr-1 h-4 w-4" /> Kompres
   374→   374→            </Button>
   375→   375→            <Button
   376→   376→              variant="outline"
   377→   377→              size="sm"
   378→   378→              className="h-8 border-zinc-800 bg-zinc-950"
   379→   379→              disabled={sel.size === 0 || busy}
   380→   380→              onClick={() => { setInputValue(''); setDialog({ kind: 'moveTo' }) }}
   381→   381→            >
   382→   382→              <FolderInput className="mr-1 h-4 w-4" /> Move
   383→   383→            </Button>
   384→   384→            <Button
   385→   385→              variant="outline"
   386→   386→              size="sm"
   387→   387→              className="h-8 border-zinc-800 bg-zinc-950 text-red-300 hover:bg-red-950/40 hover:text-red-200"
   388→   388→              disabled={sel.size === 0 || busy}
   389→   389→              onClick={() => setDialog({ kind: 'delSel' })}
   390→   390→            >
   391→   391→              <Trash2 className="mr-1 h-4 w-4" /> Hapus
   392→   392→            </Button>
   393→   393→            <Button
   394→   394→              variant="outline"
   395→   395→              size="sm"
   396→   396→              className="h-8 border-zinc-800 bg-zinc-950 text-amber-300 hover:bg-amber-950/30 hover:text-amber-200"
   397→   397→              disabled={selZip.length === 0 || busy}
   398→   398→              onClick={() => setDialog({ kind: 'extract' })}
   399→   399→            >
   400→   400→              <FolderOutput className="mr-1 h-4 w-4" /> Extract
   401→   401→            </Button>
   402→   402→            <Button
   403→   403→              variant="outline"
   404→   404→              size="sm"
   405→   405→              className="h-8 border-zinc-800 bg-zinc-950"
   406→   406→              onClick={() => enterSelect('none')}
   407→   407→            >
   408→   408→              Batal
   409→   409→            </Button>
   410→   410→          </div>
   411→   411→        </div>
   412→   412→      )}
   413→   413→
   414→   414→      <ScrollArea className="h-[55vh] min-h-[320px] rounded-lg border border-zinc-800 bg-zinc-900/40">
   415→   415→        {loading ? (

   417→   417→            <Loader2 className="h-6 w-6 animate-spin text-emerald-500" />
   418→   418→          </div>
   419→   419→        ) : entries.length === 0 ? (
   420→   420→          <p className="py-16 text-center text-sm text-zinc-600">Folder kosong — upload atau bikin file dulu.</p>
